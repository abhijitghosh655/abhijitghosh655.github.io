<?php
header("Access-Control-Allow-Origin: *");
header("Content-Type: application/json; charset=UTF-8");
header("Access-Control-Allow-Methods: GET, POST, PUT, DELETE");
header("Access-Control-Allow-Headers: Content-Type, Access-Control-Allow-Headers, Authorization, X-Requested-With");

include_once 'db.php';

$database = new Database();
$db = $database->getConnection();

$method = $_SERVER['REQUEST_METHOD'];

switch($method) {
    case 'GET':
        // Get all products or specific product
        if(isset($_GET['id'])) {
            getProduct($db, $_GET['id']);
        } else if(isset($_GET['element'])) {
            getProductsByElement($db, $_GET['element']);
        } else {
            getProducts($db);
        }
        break;
        
    case 'POST':
        // Add new product
        addProduct($db);
        break;
        
    case 'PUT':
        // Update product
        updateProduct($db);
        break;
        
    case 'DELETE':
        // Delete product
        deleteProduct($db);
        break;
}

function getProducts($db) {
    $query = "SELECT p.*, e.name as element_name FROM products p 
              LEFT JOIN elements e ON p.element_symbol = e.symbol 
              ORDER BY p.created_at DESC";
    
    $stmt = $db->prepare($query);
    $stmt->execute();
    
    $products = array();
    while ($row = $stmt->fetch(PDO::FETCH_ASSOC)) {
        $products[] = $row;
    }
    
    echo json_encode($products);
}

function getProductsByElement($db, $element) {
    $query = "SELECT p.*, e.name as element_name FROM products p 
              LEFT JOIN elements e ON p.element_symbol = e.symbol 
              WHERE p.element_symbol = :element 
              ORDER BY p.price ASC";
    
    $stmt = $db->prepare($query);
    $stmt->bindParam(':element', $element);
    $stmt->execute();
    
    $products = array();
    while ($row = $stmt->fetch(PDO::FETCH_ASSOC)) {
        $products[] = $row;
    }
    
    echo json_encode($products);
}

function getProduct($db, $id) {
    $query = "SELECT p.*, e.name as element_name FROM products p 
              LEFT JOIN elements e ON p.element_symbol = e.symbol 
              WHERE p.id = :id";
    
    $stmt = $db->prepare($query);
    $stmt->bindParam(':id', $id);
    $stmt->execute();
    
    $product = $stmt->fetch(PDO::FETCH_ASSOC);
    echo json_encode($product);
}

function addProduct($db) {
    $data = json_decode(file_get_contents("php://input"));
    
    $query = "INSERT INTO products 
              SET name=:name, description=:description, price=:price, 
                  element_symbol=:element_symbol, purity=:purity, 
                  category=:category, image_url=:image_url, stock_quantity=:stock_quantity";
    
    $stmt = $db->prepare($query);
    
    $stmt->bindParam(':name', $data->name);
    $stmt->bindParam(':description', $data->description);
    $stmt->bindParam(':price', $data->price);
    $stmt->bindParam(':element_symbol', $data->element_symbol);
    $stmt->bindParam(':purity', $data->purity);
    $stmt->bindParam(':category', $data->category);
    $stmt->bindParam(':image_url', $data->image_url);
    $stmt->bindParam(':stock_quantity', $data->stock_quantity);
    
    if($stmt->execute()) {
        echo json_encode(array("message" => "Product added successfully."));
    } else {
        echo json_encode(array("message" => "Unable to add product."));
    }
}

function updateProduct($db) {
    $data = json_decode(file_get_contents("php://input"));
    
    $query = "UPDATE products 
              SET name=:name, description=:description, price=:price, 
                  element_symbol=:element_symbol, purity=:purity, 
                  category=:category, image_url=:image_url, stock_quantity=:stock_quantity
              WHERE id=:id";
    
    $stmt = $db->prepare($query);
    
    $stmt->bindParam(':id', $data->id);
    $stmt->bindParam(':name', $data->name);
    $stmt->bindParam(':description', $data->description);
    $stmt->bindParam(':price', $data->price);
    $stmt->bindParam(':element_symbol', $data->element_symbol);
    $stmt->bindParam(':purity', $data->purity);
    $stmt->bindParam(':category', $data->category);
    $stmt->bindParam(':image_url', $data->image_url);
    $stmt->bindParam(':stock_quantity', $data->stock_quantity);
    
    if($stmt->execute()) {
        echo json_encode(array("message" => "Product updated successfully."));
    } else {
        echo json_encode(array("message" => "Unable to update product."));
    }
}

function deleteProduct($db) {
    $data = json_decode(file_get_contents("php://input"));
    
    $query = "DELETE FROM products WHERE id=:id";
    $stmt = $db->prepare($query);
    $stmt->bindParam(':id', $data->id);
    
    if($stmt->execute()) {
        echo json_encode(array("message" => "Product deleted successfully."));
    } else {
        echo json_encode(array("message" => "Unable to delete product."));
    }
}
?>
